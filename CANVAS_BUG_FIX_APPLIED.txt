function load3DModel(filename) {
  const viewer = document.getElementById("viewer-3d");

  // ❌ PROBLEM: This replaces EVERYTHING including the canvas!

    <div>Loading 3D Model...</div>
  `;

  // ❌ Now when we try to find canvas, it's gone!
  if (!initThreeJSScene()) {  // ← Canvas not found here!
    // This error path gets triggered
  }

  // ❌ Three.js never initializes
}


After Fix (WORKING):
───────────────────
function load3DModel(filename) {
  const viewer = document.getElementById("viewer-3d");

  // ✅ SOLUTION: Preserve canvas, add content after
  viewer.innerHTML = '<canvas id="three-canvas"></canvas>';

  // ✅ Now canvas exists and Three.js can initialize
  if (!initThreeJSScene()) {  // ← Canvas found!
    // Error handling (rarely triggered now)
    return;
  }

  // ✅ Three.js renders on canvas

  // ✅ Show loading overlay (doesn't destroy canvas)
  const loadingDiv = document.createElement("div");
  loadingDiv.style.position = "absolute";
  // ... loading indicator displayed on top
  viewer.appendChild(loadingDiv);

  // ✅ Model loads successfully
}
