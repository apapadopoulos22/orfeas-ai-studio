TROUBLESHOOTING
═══════════════════════════════════════════════════════════════════════════════

PROBLEM: Docker image won't build
FIX:
  1. Check Docker is running: docker ps
  2. Verify Dockerfile: ls -la Dockerfile
  3. Check requirements.txt: ls -la backend/requirements.txt
  4. Try rebuild: docker build --no-cache -t bob-ai:latest .

PROBLEM: Port 5000 already in use
FIX:
  1. Check what's using port: lsof -i :5000
  2. Kill process: kill -9 <PID>
  3. Or change port in docker-compose.yml

PROBLEM: Container exits immediately
FIX:
  1. Check logs: docker-compose logs api
  2. Check health: curl http://localhost:5000/health
  3. Verify volumes: docker-compose config | grep volumes

PROBLEM: Health check failing
FIX:
  1. Check service running: docker-compose ps
  2. Check logs: docker-compose logs api
  3. Check port mapping: docker port <container_id>
  4. Test directly: curl -v http://localhost:5000/health

PROBLEM: Volumes not mounting
FIX:
  1. Verify volume config: docker-compose config | grep volumes
  2. Check permissions: ls -la ./models ./outputs ./logs
  3. Recreate volumes: docker-compose down -v && docker-compose up -d

DOCKER COMPOSE FILE REFERENCE

version: '3.9'

services:
  api:
    image: bob-ai:latest
    container_name: bob-ai-api
    ports:
      - "5000:5000"                  # API port mapping
    environment:
      - FLASK_ENV=production
      - PYTHONUNBUFFERED=1
      - DEVICE=cuda
    volumes:
      - ./models:/app/models          # Mount models directory
      - ./outputs:/app/outputs        # Mount outputs directory
      - ./logs:/app/logs             # Mount logs directory
      - ./uploads:/app/uploads       # Mount uploads directory
    networks:
      - bob-ai-network
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:5000/health"]
      interval: 30s
      timeout: 10s
      retries: 3

  monitoring:
    image: bob-ai:latest
    container_name: bob-ai-monitoring
    command: python -u backend/phase4_monitoring_app.py
    ports:
      - "8000:8000"                  # Monitoring port mapping
    volumes:
      - ./logs:/app/logs             # Mount logs directory
    networks:
      - bob-ai-network
    depends_on:
      api:
        condition: service_healthy
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8000/health"]
      interval: 30s
      timeout: 10s
      retries: 3

volumes:
  models:
  outputs:
  logs:
  uploads:

networks:
  bob-ai-network:
    driver: bridge
